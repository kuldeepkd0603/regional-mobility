class User(Base):
    __tablename__ = "users"

    id: Mapped[int] = mapped_column(Integer, primary_key=True, index=True)

    full_name: Mapped[str] = mapped_column(String(100), nullable=False)
    phone: Mapped[str] = mapped_column(String(20), unique=True, index=True, nullable=False)
    email: Mapped[str | None] = mapped_column(String(150), unique=True, nullable=True)

    role: Mapped[UserRole] = mapped_column(
        Enum(UserRole, name="user_role"),
        default=UserRole.PASSENGER,
        nullable=False,
    )

    is_active: Mapped[bool] = mapped_column(Boolean, default=True)
    created_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), default=datetime.utcnow)

    driver_profile: Mapped["Driver | None"] = relationship(
        back_populates="user",
        uselist=False,
    )
    
    agency_profile: Mapped["Agency | None"] = relationship(
    back_populates="user",
    uselist=False,
)
    
    hashed_password: Mapped[str | None] = mapped_column(
    String(255),
    nullable=True
    )

class BookingCreate(BaseModel):
    passenger_id: int

    trip_type: TripType
    preferred_vehicle_type: str = Field(
    pattern="^(dzire|eeco|ertiga|innova|urbania)$"
    )

    pickup_city: str = Field(min_length=2, max_length=100)
    drop_city: str = Field(min_length=2, max_length=100)

    pickup_address: str = Field(min_length=3)
    drop_address: str = Field(min_length=3)

    distance_km: float = Field(gt=0, le=2000)

    scheduled_at: datetime
    return_at: datetime | None = None

    passenger_count: int = Field(default=1, ge=1, le=50)
    special_instructions: str | None = None
    pickup_latitude: float | None = None
    pickup_longitude: float | None = None

    @model_validator(mode="after")
    def validate_return_trip(self):
        if self.trip_type == TripType.ROUND_TRIP:
            if self.return_at is None:
                raise ValueError(
                    "Return date and time are required for round trips"
                )

            if self.return_at <= self.scheduled_at:
                raise ValueError(
                    "Return time must be after pickup time"
                )

        elif self.return_at is not None:
            raise ValueError(
                "Return time is only allowed for round trips"
            )

        return self

@router.get(

def get_assigned_driver_location(
    booking_id: int,
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_roles(UserRole.PASSENGER, UserRole.ADMIN)
    )
):
    booking = db.get(Booking, booking_id)

    if not booking:
        raise HTTPException(

    # Passenger can track only their own booking.
    if (

        and booking.passenger_id != current_user.id
    ):
        raise HTTPException(
            status_code=403,

        BookingStatus.DRIVER_ASSIGNED,
        BookingStatus.ONGOING
    ]:
        raise HTTPException(

            detail="Driver tracking is available only for assigned or ongoing trips"
        )

    if not booking.driver_id:
        raise HTTPException(
            status_code=404,
            detail="No driver assigned"
        )

    driver = db.get(Driver, booking.driver_id)

    if not driver:
        raise HTTPException(
            status_code=404,
            detail="Driver not found"
        )

    if driver.latitude is None or driver.longitude is None:
        raise HTTPException(
            status_code=404,
            detail="Driver location is not available yet"
        )

    if driver.location_updated_at is None:
        raise HTTPException(
            status_code=404,
            detail="Driver has not shared their location"

    updated_at = driver.location_updated_at

    # Handle timezone-naive datetime.
    if updated_at.tzinfo is None:
        updated_at = updated_at.replace(tzinfo=timezone.utc)

        )

@router.get(
    "/nearby",
    response_model=list[NearbyDriverResponse]
)
def find_nearby_drivers(
    latitude: float = Query(ge=-90, le=90),
    longitude: float = Query(ge=-180, le=180),
    radius_km: float = Query(default=10, gt=0, le=100),
    passenger_count: int = Query(default=1, ge=1, le=50),
    vehicle_type: str | None = Query(default=None),

    db: Session = Depends(get_db),

    current_user: User = Depends(
        require_roles(UserRole.PASSENGER, UserRole.ADMIN)
    )
):

    # Validate that coordinates are within valid geographic ranges.
    if not (-90 <= latitude <= 90):
        raise HTTPException(
            status_code=400,
            detail="Invalid latitude"
        )

    if not (-180 <= longitude <= 180):
        raise HTTPException(
            status_code=400,
            detail="Invalid longitude"
        )

    # SQLAlchemy text() uses :parameter syntax.
    sql = text("""
        SELECT
            d.id AS driver_id,
            u.full_name AS driver_name,

            v.id AS vehicle_id,
            v.vehicle_name,
            v.vehicle_type,
            v.registration_number,
            v.seating_capacity,

            d.rating,
            d.latitude,
            d.longitude,

            extensions.ST_Distance(
                extensions.ST_SetSRID(
                    extensions.ST_MakePoint(
                        d.longitude,
                        d.latitude
                    ),
                    4326
                )::extensions.geography,

                extensions.ST_SetSRID(
                    extensions.ST_MakePoint(
                        :longitude,
                        :latitude
                    ),
                    4326
                )::extensions.geography
            ) / 1000.0 AS distance_km

        FROM drivers d

        JOIN users u
            ON u.id = d.user_id

        JOIN vehicles v
            ON v.driver_id = d.id

        WHERE

            d.is_available = TRUE

            AND d.verification_status = 'VERIFIED'

            AND u.is_active = TRUE

            AND v.is_active = TRUE

            AND d.latitude IS NOT NULL

            AND d.longitude IS NOT NULL

            -- Driver location must be updated within 5 minutes.
            AND d.location_updated_at >=
                NOW() - INTERVAL '5 minutes'

            -- Vehicle must have enough seats.
            AND v.seating_capacity >= :passenger_count

            -- Optional vehicle type filter.
            AND (
                CAST(:vehicle_type AS TEXT) IS NULL
                OR LOWER(v.vehicle_type) =
                   LOWER(CAST(:vehicle_type AS TEXT))
            )

            -- Search drivers within the selected radius.
            AND extensions.ST_DWithin(

                extensions.ST_SetSRID(
                    extensions.ST_MakePoint(
                        d.longitude,
                        d.latitude
                    ),
                    4326
                )::extensions.geography,

                extensions.ST_SetSRID(
                    extensions.ST_MakePoint(
                        :longitude,
                        :latitude
                    ),
                    4326
                )::extensions.geography,

                :radius_m
            )

        ORDER BY distance_km ASC

        LIMIT 50
    """)

    result = db.execute(
        sql,
        {
            "longitude": longitude,
            "latitude": latitude,
            "passenger_count": passenger_count,
            "vehicle_type": vehicle_type,
            "radius_m": radius_km * 1000,
        }
    )

    drivers = [
        dict(row)
        for row in result.mappings().all()
    ]

    return drivers

@router.post("/distance")
async def calculate_route_distance(request: DistanceRequest):
    api_key = os.getenv("GOOGLE_MAPS_API_KEY")

    if not api_key:
        raise HTTPException(
            status_code=500,
            detail="Google Routes API key is not configured",
        )

    payload = {
        "origin": {
            "location": {
                "latLng": {
                    "latitude": request.pickup_latitude,
                    "longitude": request.pickup_longitude,
                }
            }
        },
        "destination": {
            "location": {
                "latLng": {
                    "latitude": request.drop_latitude,
                    "longitude": request.drop_longitude,
                }
            }
        },
        "travelMode": "DRIVE",
        "routingPreference": "TRAFFIC_AWARE",
        "computeAlternativeRoutes": False,
    }

    headers = {
        "Content-Type": "application/json",
        "X-Goog-Api-Key": api_key,
        "X-Goog-FieldMask": (
            "routes.distanceMeters,"
            "routes.duration,"
            "routes.polyline.encodedPolyline"
        ),
    }

    try:
        async with httpx.AsyncClient(timeout=20.0) as client:
            response = await client.post(
                GOOGLE_ROUTES_URL,
                json=payload,
                headers=headers,
            )

        if response.status_code != 200:
            print("Google API Status:", response.status_code)
            print("Google API Response:", response.text)

            raise HTTPException(
                status_code=502,
                detail={
                    "google_status": response.status_code,
                    "google_error": response.json(),
                },
            )

        data = response.json()
        routes = data.get("routes", [])

        if not routes:
            raise HTTPException(
                status_code=404,
                detail="No driving route found between these locations",
            )

        route = routes[0]
        distance_meters = route["distanceMeters"]
        distance_km = round(distance_meters / 1000, 2)

        duration_text = route.get("duration", "0s")
        duration_seconds = int(duration_text.removesuffix("s"))

        return {
            "distance_meters": distance_meters,
            "distance_km": distance_km,
            "duration": duration_text,
            "duration_seconds": duration_seconds,
            "polyline": route.get("polyline", {}).get(
                "encodedPolyline"
            ),
        }

    except httpx.TimeoutException:
        raise HTTPException(
            status_code=504,
            detail="Google Routes API request timed out",
        )
    except httpx.RequestError:
        raise HTTPException(
            status_code=502,
            detail="Unable to connect to Google Routes API",
        )

def normalize_datetime(value: datetime) -> datetime:
    """
    Treat timezone-naive booking times as India local time.
    Store timezone-aware values.

    if value.tzinfo is None or value.utcoffset() is None:

def get_booking_window(booking, route):
    """
    Calculate the occupied time window for a booking.

    Round trips use return_at.
    One-way trips use estimated duration plus buffer.
    Ongoing trips block further assignments until completed.
    """

    if not booking.scheduled_at:
        raise HTTPException(
            status_code=409,
            detail="Booking pickup date and time are required for assignment"
        )

    start_time = normalize_datetime(booking.scheduled_at)

    # An ongoing trip is treated as busy until it is completed.
    if booking.booking_status == BookingStatus.ONGOING:
        return (
            start_time,
            datetime.max.replace(tzinfo=timezone.utc)
        )

    # Round-trip booking
    if booking.trip_type == TripType.ROUND_TRIP:

        if not booking.return_at:
            raise HTTPException(
                status_code=409,
                detail="Round-trip booking is missing its return time"
            )

        end_time = normalize_datetime(booking.return_at)

        if end_time <= start_time:
            raise HTTPException(
                status_code=409,
                detail="Invalid round-trip schedule"
            )

        return start_time, end_time

    # One-way booking duration estimation
    if not route or not route.distance_km or route.distance_km <= 0:
        raise HTTPException(
            status_code=409,
            detail="Valid route distance is required for scheduling"
        )

    average_speed = AVERAGE_SPEED_KMPH.get(booking.trip_type)

    if not average_speed:
        raise HTTPException(
            status_code=409,
            detail="Unable to estimate this trip duration"
        )

    estimated_hours = route.distance_km / average_speed

    duration = (
        timedelta(hours=estimated_hours)
        + timedelta(minutes=BUFFER_MINUTES)
    )

    end_time = start_time + duration

    return start_time, end_time


def check_schedule_conflicts(
    db: Session,
    booking: Booking,
    driver_id: int,
    vehicle_id: int
):
    """
    Reject overlapping bookings for the same driver or vehicle.
    """

    new_route = db.get(RideRoute, booking.route_id)

    new_start, new_end = get_booking_window(
        booking,
        new_route
    )

    existing_bookings = (
        db.query(Booking)
        .filter(
            Booking.id != booking.id,
            Booking.booking_status.in_(
                ACTIVE_BOOKING_STATUSES
            ),
            or_(
                Booking.driver_id == driver_id,
                Booking.vehicle_id == vehicle_id
            )
        )
        .with_for_update()
        .all()
    )

    for existing in existing_bookings:

        existing_route = db.get(
            RideRoute,
            existing.route_id
        )

        existing_start, existing_end = get_booking_window(
            existing,
            existing_route
        )

        # Standard interval overlap check
        if (
            existing_start < new_end
            and existing_end > new_start
        ):

            if existing.driver_id == driver_id:
                detail = (
                    f"Driver already has an overlapping booking "
                    f"(Booking ID: {existing.id})"
                )
            else:
                detail = (
                    f"Vehicle already has an overlapping booking "
                    f"(Booking ID: {existing.id})"
                )

            raise HTTPException(
                status_code=409,
                detail=detail
            )

@router.post(

    response_model=BookingResponse,
    status_code=201
)
def create_booking(
    data: BookingCreate,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user)
):

    # Only passengers can create bookings
    if current_user.role != UserRole.PASSENGER:
        raise HTTPException(
            status_code=403,
            detail="Only passengers can create bookings"
        )

    # Prevent creating bookings for another passenger
    if data.passenger_id != current_user.id:
        raise HTTPException(
            status_code=403,
            detail="You cannot create a booking for another passenger"
        )

    passenger = db.query(User).filter(
        User.id == current_user.id,
        User.role == UserRole.PASSENGER,
        User.is_active.is_(True)
    ).first()

    if not passenger:
        raise HTTPException(
            status_code=404,
            detail="Active passenger not found"
        )

    try:

        route = RideRoute(
            pickup_city=data.pickup_city,
            drop_city=data.drop_city,
            pickup_address=data.pickup_address,
            drop_address=data.drop_address,
            distance_km=data.distance_km,
            pickup_latitude=data.pickup_latitude,
            pickup_longitude=data.pickup_longitude,
        )

        db.add(route)
        db.flush()

        booking = Booking(
            passenger_id=current_user.id,
            route_id=route.id,
            trip_type=data.trip_type,
            scheduled_at=normalize_datetime(data.scheduled_at),
            return_at=(
                normalize_datetime(data.return_at)
                if data.return_at
                else None
            ),
            passenger_count=data.passenger_count,
            special_instructions=data.special_instructions,
            preferred_vehicle_type=data.preferred_vehicle_type,

            estimated_fare=calculate_fare(
                data.distance_km,
                data.trip_type,
                data.preferred_vehicle_type
            ),
            booking_status=BookingStatus.PENDING,
        )

        db.add(booking)
        db.flush()

        # Notify all active administrators
        admins = db.query(User).filter(
            User.role == UserRole.ADMIN,
            User.is_active.is_(True)
        ).all()

        for admin in admins:
            create_notification(
                db=db,
                user_id=admin.id,
                title="New Booking Received",
                message=(
                    f"New booking #{booking.id} "
                    f"from {data.pickup_city} to {data.drop_city}"
                ),
                notification_type="new_booking",
                booking_id=booking.id
            )

        db.commit()
        db.refresh(booking)

        return booking

    except IntegrityError:
        db.rollback()

        raise HTTPException(
            status_code=409,
            detail="Unable to create booking due to a database conflict"
        )

    except Exception:
        db.rollback()
        raise

@router.get(
    "/",
    response_model=list[BookingResponse]
)
def get_bookings(
    passenger_id: int | None = Query(default=None),
    driver_id: int | None = Query(default=None),
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user)
):

    query = db.query(Booking)

    # Passenger can only see their own bookings
    if current_user.role == UserRole.PASSENGER:

        if passenger_id is not None and passenger_id != current_user.id:
            raise HTTPException(
                status_code=403,
                detail="You cannot access another passenger's bookings"
            )

        if driver_id is not None:
            raise HTTPException(
                status_code=403,
                detail="Passenger cannot filter bookings by driver"
            )

        query = query.filter(
            Booking.passenger_id == current_user.id
        )

    # Driver can only see assigned bookings
    elif current_user.role == UserRole.DRIVER:

        if passenger_id is not None or driver_id is not None:
            raise HTTPException(
                status_code=403,
                detail="Drivers cannot use these booking filters"
            )

        driver = db.query(Driver).filter(
            Driver.user_id == current_user.id
        ).first()

        if not driver:
            raise HTTPException(
                status_code=404,
                detail="Driver profile not found"
            )

        query = query.filter(
            Booking.driver_id == driver.id
        )

    # Admin can view all bookings
    elif current_user.role == UserRole.ADMIN:

        if passenger_id is not None:
            query = query.filter(
                Booking.passenger_id == passenger_id
            )

        if driver_id is not None:
            query = query.filter(
                Booking.driver_id == driver_id
            )
            
        # Agency can view bookings assigned to its own drivers
    elif current_user.role == UserRole.AGENCY:

        if passenger_id is not None or driver_id is not None:
            raise HTTPException(
                status_code=403,
                detail="Agency cannot use these booking filters"
            )

        query = query.join(
            Driver,
            Booking.driver_id == Driver.id
        ).filter(
            Driver.agency_id == (
                db.query(Agency.id)
                .filter(Agency.user_id == current_user.id)
                .scalar_subquery()
            )
        )
        
        # Agency can view only its own drivers' bookings
    elif current_user.role == UserRole.AGENCY:

        driver = db.query(Driver).filter(
            Driver.id == booking.driver_id,
            Driver.agency_id == (
                db.query(Agency.id)
                .filter(Agency.user_id == current_user.id)
                .scalar_subquery()
            )
        ).first()

        if not driver:
            raise HTTPException(
                status_code=403,
                detail="This booking does not belong to your agency"
            )

    else:
        raise HTTPException(
            status_code=403,
            detail="You are not authorized to view bookings"
        )
        

    return query.order_by(
        Booking.id.desc()
    ).limit(100).all()

        raise HTTPException(

        )

        raise HTTPException(
            status_code=403,
            detail="Driver account is inactive"
        )

    driver = db.query(Driver).filter(
        Driver.user_id == current_user.id
    ).first()

    if not driver:
        raise HTTPException(404, "Driver profile not found")

        raise HTTPException(
            status_code=403,
            detail="Driver is not verified"
        )

    if not driver.is_available:
        raise HTTPException(
            status_code=403,
            detail="Go online to view available requests"
        )

    # Only show requests that can be served by one of
    # this driver's active vehicles.
    active_vehicles = db.query(Vehicle).filter(
        Vehicle.driver_id == driver.id,
        Vehicle.is_active.is_(True)
    ).all()

    if not active_vehicles:
        return []

    max_capacity = max(
        vehicle.seating_capacity for vehicle in active_vehicles
    )

    requests = (
        db.query(Booking)
        .filter(
            Booking.booking_status == BookingStatus.PENDING,
            Booking.driver_id.is_(None),
            Booking.passenger_count <= max_capacity
        )
        .order_by(Booking.scheduled_at.asc(), Booking.id.desc())
        .limit(100)
        .all()
    )

    return requests

@router.patch(
    "/{booking_id}/assign",
    response_model=BookingResponse
)
def assign_driver(
    booking_id: int,
    data: BookingAssignment,
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_roles(UserRole.ADMIN)
    )
):
    # Lock booking
    booking = (
        db.query(Booking)
        .filter(Booking.id == booking_id)
        .with_for_update()
        .first()
    )

    if not booking:
        raise HTTPException(404, "Booking not found")

    if booking.booking_status != BookingStatus.PENDING:
        raise HTTPException(
            409,
            "Only pending bookings can be assigned"
        )

    if not booking.scheduled_at:
        raise HTTPException(
            409,
            "Booking pickup date and time are required"
        )

    # Lock driver
    driver = (
        db.query(Driver)
        .filter(Driver.id == data.driver_id)
        .with_for_update()
        .first()
    )

    if not driver:
        raise HTTPException(404, "Driver not found")

    driver_user = db.get(User, driver.user_id)

    if not driver_user or not driver_user.is_active:
        raise HTTPException(403, "Driver account is inactive")

    if driver.verification_status != VerificationStatus.VERIFIED:
        raise HTTPException(403, "Driver is not verified")

    if not driver.is_available:
        raise HTTPException(
            409,
            "Driver is offline or unavailable"
        )

    # Lock vehicle
    vehicle = (
        db.query(Vehicle)
        .filter(
            Vehicle.id == data.vehicle_id,
            Vehicle.driver_id == driver.id,
            Vehicle.is_active.is_(True)
        )
        .with_for_update()
        .first()
    )

    if not vehicle:
        raise HTTPException(
            400,
            "Vehicle is inactive or does not belong to this driver"
        )

    if vehicle.seating_capacity < booking.passenger_count:
        raise HTTPException(
            400,
            "Vehicle does not have enough passenger capacity"
        )

    # Check schedule conflicts
    check_schedule_conflicts(
        db=db,
        booking=booking,
        driver_id=driver.id,
        vehicle_id=vehicle.id
    )

    # Assign driver and vehicle
    booking.driver_id = driver.id
    booking.vehicle_id = vehicle.id
    booking.booking_status = BookingStatus.DRIVER_ASSIGNED

    # Notify passenger
    create_notification(
        db=db,
        user_id=booking.passenger_id,
        title="Driver Assigned",
        message=(
            f"Your booking #{booking.id} has been assigned "
            f"to driver {driver_user.full_name}"
        ),
        notification_type="driver_assigned",
        booking_id=booking.id
    )

    # Notify driver
    create_notification(
        db=db,
        user_id=driver.user_id,
        title="New Trip Assigned",
        message=f"You have been assigned booking #{booking.id}",
        notification_type="trip_assigned",
        booking_id=booking.id
    )

    try:
        db.commit()
        db.refresh(booking)
        return booking

    except IntegrityError:
        db.rollback()
        raise HTTPException(
            409,
            "Booking assignment conflict"
        )

@router.patch(
    "/{booking_id}/status",
    response_model=BookingResponse
)
def update_booking_status(
    booking_id: int,
    data: BookingStatusUpdate,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user)
):
    booking = (
        db.query(Booking)
        .filter(Booking.id == booking_id)
        .with_for_update()
        .first()
    )

    if not booking:
        raise HTTPException(404, detail="Booking not found")

    driver = None

    # PASSENGER PERMISSIONS
    if current_user.role == UserRole.PASSENGER:

        if booking.passenger_id != current_user.id:
            raise HTTPException(
                403,
                detail="You cannot update this booking"
            )

        if data.booking_status != BookingStatus.CANCELLED:
            raise HTTPException(
                403,
                detail="Passengers can only cancel bookings"
            )

        if booking.booking_status != BookingStatus.PENDING:
            raise HTTPException(
                400,
                detail="Passenger can only cancel pending bookings"
            )

    # DRIVER PERMISSIONS
    elif current_user.role == UserRole.DRIVER:

        driver = (
            db.query(Driver)
            .filter(Driver.user_id == current_user.id)
            .with_for_update()
            .first()
        )

        if not driver or booking.driver_id != driver.id:
            raise HTTPException(
                403,
                detail="This booking is not assigned to you"
            )

        allowed_driver_actions = {
            BookingStatus.ONGOING,
            BookingStatus.COMPLETED,
            BookingStatus.CANCELLED,
        }

        if data.booking_status not in allowed_driver_actions:
            raise HTTPException(
                403,
                detail="Driver is not allowed to perform this action"
            )

    # ADMIN PERMISSIONS
    elif current_user.role != UserRole.ADMIN:

        raise HTTPException(
            403,
            detail="You are not authorized to update bookings"
        )

    # VALIDATE STATUS TRANSITIONS
    allowed_transitions = {
        BookingStatus.PENDING: {
            BookingStatus.CANCELLED,
        },
        BookingStatus.ACCEPTED: {
            BookingStatus.CANCELLED,
        },
        BookingStatus.DRIVER_ASSIGNED: {
            BookingStatus.ONGOING,
            BookingStatus.CANCELLED,
        },
        BookingStatus.ONGOING: {
            BookingStatus.COMPLETED,
        },
    }

    if data.booking_status not in allowed_transitions.get(
        booking.booking_status, set()
    ):
        raise HTTPException(
            409,
            detail=(
                f"Cannot change booking from "
                f"{booking.booking_status.value} to "
                f"{data.booking_status.value}"
            )
        )

    old_status = booking.booking_status
    booking.booking_status = data.booking_status

    # CANCELLATION WORKFLOW
    if data.booking_status == BookingStatus.CANCELLED:

        # Lock and cancel all pending driver requests
        pending_requests = (
            db.query(DriverBookingRequest)
            .filter(
                DriverBookingRequest.booking_id == booking.id,
                DriverBookingRequest.status == "pending",
            )
            .with_for_update()
            .all()
        )

        for request in pending_requests:
            request.status = "cancelled"
            request.responded_at = datetime.now(timezone.utc)

        # Notify passenger when driver or admin cancels
        if current_user.role in {
            UserRole.DRIVER,
            UserRole.ADMIN,
        }:
            create_notification(
                db=db,
                user_id=booking.passenger_id,
                title="Booking Cancelled",
                message=f"Booking #{booking.id} has been cancelled.",
                notification_type="booking_cancelled",
                booking_id=booking.id,
            )

        # Notify assigned driver when admin cancels
        if (
            current_user.role == UserRole.ADMIN
            and booking.driver_id
        ):
            assigned_driver = db.get(
                Driver,
                booking.driver_id
            )

            if assigned_driver:
                create_notification(
                    db=db,
                    user_id=assigned_driver.user_id,
                    title="Booking Cancelled",
                    message=f"Booking #{booking.id} has been cancelled by admin.",
                    notification_type="booking_cancelled",
                    booking_id=booking.id,
                )

        # Notify drivers whose pending requests were cancelled
        for request in pending_requests:
            requested_driver = db.get(
                Driver,
                request.driver_id
            )

            if requested_driver:
                create_notification(
                    db=db,
                    user_id=requested_driver.user_id,
                    title="Booking Request Cancelled",
                    message=f"Booking request #{booking.id} is no longer available.",
                    notification_type="booking_cancelled",
                    booking_id=booking.id,
                )

        # Restore assigned driver's availability when safe
        if (
            old_status == BookingStatus.DRIVER_ASSIGNED
            and booking.driver_id
        ):
            assigned_driver = (
                db.query(Driver)
                .filter(
                    Driver.id == booking.driver_id
                )
                .with_for_update()
                .first()
            )

            if assigned_driver:

                other_active_booking = (
                    db.query(Booking.id)
                    .filter(
                        Booking.driver_id == assigned_driver.id,
                        Booking.id != booking.id,
                        Booking.booking_status.in_([
                            BookingStatus.DRIVER_ASSIGNED,
                            BookingStatus.ONGOING,
                        ]),
                    )
                    .first()
                )

                if (
                    not other_active_booking
                    and assigned_driver.verification_status
                    == VerificationStatus.VERIFIED
                ):
                    assigned_driver.is_available = True

    db.commit()
    db.refresh(booking)

    return booking

@router.patch(
    "/{booking_id}/accept",
    response_model=BookingResponse
)
def accept_booking(
    booking_id: int,
    data: DriverBookingAccept,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user)
):
    if current_user.role != UserRole.DRIVER:
        raise HTTPException(
            status_code=403,
            detail="Only drivers can accept ride requests"
        )

    if not current_user.is_active:
        raise HTTPException(
            status_code=403,
            detail="Driver account is inactive"
        )

    # Lock the booking so two drivers cannot accept it together.
    booking = (
        db.query(Booking)
        .filter(Booking.id == booking_id)
        .with_for_update()
        .first()
    )

    if not booking:
        raise HTTPException(404, "Booking not found")

    if (
        booking.booking_status != BookingStatus.PENDING
        or booking.driver_id is not None
    ):
        raise HTTPException(
            status_code=409,
            detail="This booking is no longer available"
        )

    driver = (
        db.query(Driver)
        .filter(Driver.user_id == current_user.id)
        .with_for_update()
        .first()
    )

    if not driver:
        raise HTTPException(404, "Driver profile not found")

    if driver.verification_status != VerificationStatus.VERIFIED:
        raise HTTPException(
            status_code=403,
            detail="Driver is not verified"
        )

    if not driver.is_available:
        raise HTTPException(
            status_code=403,
            detail="You must be online to accept a ride"
        )

    # Lock and validate the driver's selected vehicle.
    vehicle = (
        db.query(Vehicle)
        .filter(
            Vehicle.id == data.vehicle_id,
            Vehicle.driver_id == driver.id,
            Vehicle.is_active.is_(True)
        )
        .with_for_update()
        .first()
    )

    if not vehicle:
        raise HTTPException(
            status_code=400,
            detail="Selected vehicle is inactive or does not belong to you"
        )

    if vehicle.seating_capacity < booking.passenger_count:
        raise HTTPException(
            status_code=400,
            detail="Vehicle does not have enough passenger capacity"
        )

    # Reuse the existing schedule conflict checker.
    check_schedule_conflicts(
        db=db,
        booking=booking,
        driver_id=driver.id,
        vehicle_id=vehicle.id
    )

    booking.driver_id = driver.id
    booking.vehicle_id = vehicle.id
    booking.booking_status = BookingStatus.DRIVER_ASSIGNED

    create_notification(
        db=db,
        user_id=booking.passenger_id,
        title="Driver Accepted Your Booking",
        message=(
            f"Your booking #{booking.id} has been accepted "
            f"by driver {current_user.full_name}."
        ),
        notification_type="driver_assigned",
        booking_id=booking.id
    )

    create_notification(
        db=db,
        user_id=current_user.id,
        title="Ride Accepted",
        message=f"You have accepted booking #{booking.id}.",
        notification_type="trip_assigned",
        booking_id=booking.id
    )

    try:
        db.commit()
        db.refresh(booking)
        return booking

    except IntegrityError:
        db.rollback()
        raise HTTPException(
            status_code=409,
            detail="Booking acceptance conflict. Please try again."
        )

@router.patch("/driver-requests/{request_id}/accept")
def accept_driver_booking_request(
    request_id: int,
    request_data: DriverBookingRequestAction,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    if current_user.role != UserRole.DRIVER:
        raise HTTPException(status_code=403, detail="Only drivers can accept requests")

    driver = db.query(Driver).filter(
        Driver.user_id == current_user.id
    ).first()

    if not driver:
        raise HTTPException(status_code=404, detail="Driver profile not found")

    try:
        driver_request = (
            db.query(DriverBookingRequest)
            .filter(
                DriverBookingRequest.id == request_id,
                DriverBookingRequest.driver_id == driver.id,
            )
            .with_for_update()
            .first()
        )

        if not driver_request:
            raise HTTPException(status_code=404, detail="Request not found")

        if driver_request.status != "pending":
            raise HTTPException(status_code=400, detail="Request is no longer pending")

        now = datetime.now(timezone.utc)
        expiry = driver_request.expires_at

        if expiry.tzinfo is None:
            expiry = expiry.replace(tzinfo=timezone.utc)

        if expiry <= now:
            driver_request.status = "expired"
            driver_request.responded_at = now
            db.commit()
            raise HTTPException(status_code=400, detail="Request has expired")

        booking = (
            db.query(Booking)
            .filter(Booking.id == driver_request.booking_id)
            .with_for_update()
            .first()
        )

        if not booking or booking.booking_status != BookingStatus.PENDING:
            raise HTTPException(status_code=400, detail="Booking is no longer available")

        if booking.driver_id is not None:
            raise HTTPException(status_code=400, detail="Booking already assigned")

        if not driver.is_available or driver.verification_status != VerificationStatus.VERIFIED:
            raise HTTPException(status_code=400, detail="Driver is not available or verified")

        vehicle_query = db.query(Vehicle).filter(
            Vehicle.driver_id == driver.id,
            Vehicle.is_active == True,
            Vehicle.seating_capacity >= (booking.passenger_count or 1),
        )

        if request_data.vehicle_id is not None:
            vehicle_query = vehicle_query.filter(
                Vehicle.id == request_data.vehicle_id
            )

        vehicle = vehicle_query.first()

        if not vehicle:
            raise HTTPException(
                status_code=400,
                detail="No suitable active vehicle found"
            )

        if (
            booking.preferred_vehicle_type
            and vehicle.vehicle_type.lower()
            != booking.preferred_vehicle_type.lower()
        ):
            raise HTTPException(status_code=400, detail="Vehicle type does not match")

        # Assign booking to this driver
        booking.driver_id = driver.id
        booking.vehicle_id = vehicle.id
        booking.booking_status = BookingStatus.DRIVER_ASSIGNED
        driver.is_available = False

        driver_request.status = "accepted"
        driver_request.responded_at = now

        # Cancel all other outstanding requests for this booking
        other_requests = (
            db.query(DriverBookingRequest)
            .filter(
                DriverBookingRequest.booking_id == booking.id,
                DriverBookingRequest.id != driver_request.id,
                DriverBookingRequest.status == "pending",
            )
            .with_for_update()
            .all()
        )

        for other in other_requests:
            other.status = "cancelled"
            other.responded_at = now

        create_notification(
            db=db,
            user_id=booking.passenger_id,
            booking_id=booking.id,
            title="Driver assigned",
            message="A driver has accepted your booking.",
            notification_type="booking",
        )

        db.commit()

        return {
            "success": True,
            "message": "Booking accepted successfully",
            "booking_id": booking.id,
            "driver_id": driver.id,
            "vehicle_id": vehicle.id,
            "status": "driver_assigned",
        }

    except HTTPException:
        db.rollback()
        raise
    except Exception:
        db.rollback()
        raise

# Shared Pricing Configuration
# All prices are in INR.

VEHICLE_RATES = {
    "dzire": {"base_fare": 100.0, "per_km": 12.0},
    "eeco": {"base_fare": 100.0, "per_km": 11.0},
    "ertiga": {"base_fare": 150.0, "per_km": 15.0},
    "innova": {"base_fare": 200.0, "per_km": 18.0},
    "urbania": {"base_fare": 500.0, "per_km": 25.0},
}

TRIP_MULTIPLIERS = {
    "local": 1.0,
    "intercity": 1.0,
    "round_trip": 2.0,
    "religious_tour": 1.0,
}


def get_vehicle_rate(vehicle_type: str) -> dict:
    vehicle_type = vehicle_type.lower()

    rates = VEHICLE_RATES.get(vehicle_type)

    if rates is None:
        raise ValueError(
            f"Unsupported vehicle type: {vehicle_type}"
        )

    return rates


def calculate_estimated_fare(
    vehicle_type: str,
    distance_km: float,
    trip_type: str,
) -> dict:

    rates = get_vehicle_rate(vehicle_type)

    multiplier = TRIP_MULTIPLIERS.get(trip_type)

    if multiplier is None:
        raise ValueError(f"Unsupported trip type: {trip_type}")

    chargeable_distance = distance_km * multiplier

    estimated_fare = (
        rates["base_fare"]
        + chargeable_distance * rates["per_km"]
    )

    return {
        "base_fare": rates["base_fare"],
        "per_km_rate": rates["per_km"],
        "chargeable_distance_km": chargeable_distance,
        "estimated_fare": round(estimated_fare, 2),
    }


def calculate_final_fare(
    vehicle_type: str,
    actual_distance_km: float,
) -> float:

    rates = get_vehicle_rate(vehicle_type)

    final_fare = (
        rates["base_fare"]
        + actual_distance_km * rates["per_km"]
    )

    return round(final_fare, 2)
