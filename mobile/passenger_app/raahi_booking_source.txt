  Future<void> _openMapPicker() async {
    final result = await Navigator.push<Map<String, dynamic>>(
      context,
      MaterialPageRoute(
        builder: (_) => const LocationPickerScreen(),
      ),
    );

    if (!mounted || result == null) return;

    final pickupLat =
        (result['pickup_latitude'] as num?)?.toDouble();
    final pickupLng =
        (result['pickup_longitude'] as num?)?.toDouble();
    final dropLat =
        (result['drop_latitude'] as num?)?.toDouble();
    final dropLng =
        (result['drop_longitude'] as num?)?.toDouble();

    final newPickupAddress =
        result['pickup_address']?.toString().trim() ?? '';
    final newDropAddress =
        result['drop_address']?.toString().trim() ?? '';
    final newPickupCity =
        result['pickup_city']?.toString().trim() ?? '';
    final newDropCity =
        result['drop_city']?.toString().trim() ?? '';

    if (pickupLat == null ||
        pickupLng == null ||
        dropLat == null ||
        dropLng == null) {
      return;
    }

    setState(() {
      selectedCoordinates = {
        'pickup_latitude': pickupLat,
        'pickup_longitude': pickupLng,
        'drop_latitude': dropLat,
        'drop_longitude': dropLng,
      };

      pickupAddress = newPickupAddress;
      dropAddress = newDropAddress;
      pickupCity = newPickupCity;
      dropCity = newDropCity;

      pickupController.text = pickupAddress.isNotEmpty
          ? pickupAddress
          : '${pickupLat.toStringAsFixed(5)}, ${pickupLng.toStringAsFixed(5)}';

      destinationController.text = dropAddress.isNotEmpty
          ? dropAddress
          : '${dropLat.toStringAsFixed(5)}, ${dropLng.toStringAsFixed(5)}';

      distanceSource = null;
    });
  }

  
String get _tripTypeValue =>
      selectedRide.toLowerCase().replaceAll(' ', '_');

  Future<double> _getDistanceForSelectedRoute() async {
    // Always try the actual road distance first.
    try {
      final distanceKm = await ApiService.getRouteDistanceKm(
        pickupLatitude: selectedCoordinates!['pickup_latitude']!,
        pickupLongitude: selectedCoordinates!['pickup_longitude']!,
        dropLatitude: selectedCoordinates!['drop_latitude']!,
        dropLongitude: selectedCoordinates!['drop_longitude']!,
      );

      if (distanceKm <= 0 || distanceKm > 2000) {
        throw Exception('Google returned an invalid distance.');
      }

      if (!mounted) throw Exception('Screen is no longer active.');

      setState(() {
        distanceSource = 'Google Routes';
      });
      return distanceKm;
    } catch (_) {
      // This is a temporary estimate, not the route's actual road distance.
      if (!mounted) throw Exception('Screen is no longer active.');

      setState(() {
        distanceSource = 'Temporary default';
      });
      return _fallbackDistanceKm;
    }
  }

  Future<void> _searchRides() async {
    if (selectedCoordinates == null) {
      _showMessage('Please select pickup and destination on Google Maps');
      return;
    }

    if (pickupAddress.isEmpty ||
        dropAddress.isEmpty ||
        pickupCity.isEmpty ||
        dropCity.isEmpty) {
      _showMessage(
        'Please reselect both locations so their addresses can be identified.',
      );
      return;
    }

    setState(() {
      isSearching = true;
    });

    try {
      // Calculate distance before searching for drivers.
      final distanceKm = await _getDistanceForSelectedRoute();

      if (!mounted) return;

      final drivers = await ApiService.findNearbyDrivers(
        latitude: selectedCoordinates!['pickup_latitude']!,
        longitude: selectedCoordinates!['pickup_longitude']!,
        vehicleType: selectedVehicle,
        passengerCount: 1,
        radiusKm: 10,
      );

      if (!mounted) return;

      if (drivers.isEmpty) {
        _showMessage('No available drivers found within 10 km');
        return;
      }

      Navigator.push(
        context,
        MaterialPageRoute(
          builder: (_) => AvailableRidesScreen(
            drivers: drivers,
            distanceKm: distanceKm,
            tripType: _tripTypeValue,
            vehicleModel: selectedVehicle,
            pickupAddress: pickupAddress,
            dropAddress: dropAddress,
            pickupCity: pickupCity,
            dropCity: dropCity,
          ),
        ),
      );
    } catch (e) {
      if (mounted) {
        _showMessage(e.toString().replaceFirst('Exception: ', ''));
      }
    } finally {
      if (mounted) {
        setState(() {
          isSearching = false;
        });
      }
    }
  }

class _LocationPickerScreenState
    extends State<LocationPickerScreen> {
  GoogleMapController? mapController;

  LatLng? pickupLocation;
  LatLng? dropLocation;

  bool isLoadingLocation = false;

  static const LatLng defaultLocation =
      LatLng(23.1765, 75.7885); // Ujjain

  @override
  void initState() {
    super.initState();
    _getCurrentLocation();
  }

  Future<void> _getCurrentLocation() async {
    setState(() => isLoadingLocation = true);

    try {
      bool serviceEnabled =
          await Geolocator.isLocationServiceEnabled();

      if (!serviceEnabled) {
        _showMessage('Please enable location services.');
        return;
      }

      LocationPermission permission =
          await Geolocator.checkPermission();

      if (permission == LocationPermission.denied) {
        permission = await Geolocator.requestPermission();
      }

      if (permission == LocationPermission.denied ||
          permission == LocationPermission.deniedForever) {
        _showMessage('Location permission is required to find you.');
        return;
      }

      final position = await Geolocator.getCurrentPosition();

      final location = LatLng(
        position.latitude,
        position.longitude,
      );

      if (!mounted) return;

      await mapController?.animateCamera(
        CameraUpdate.newLatLngZoom(location, 15),
      );
    } catch (e) {
      _showMessage('Unable to get current location.');
    } finally {
      if (mounted) {
        setState(() => isLoadingLocation = false);
      }
    }
  }

  void _onMapTap(LatLng position) {
    setState(() {
      if (selectingPickup) {
        pickupLocation = position;
        selectingPickup = false;
      } else {
        dropLocation = position;
      }
    });
  }

    if (!mounted) return;

    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text(message)),
    );
  }

    final result = <Marker>{};

    if (pickupLocation != null) {
      result.add(
        Marker(
          markerId: const MarkerId('pickup'),
          position: pickupLocation!,
          infoWindow: const InfoWindow(title: 'Pickup'),
          icon: BitmapDescriptor.defaultMarkerWithHue(

          icon: BitmapDescriptor.defaultMarkerWithHue(

  }

  Future<void> _confirmLocations() async {
    if (pickupLocation == null || dropLocation == null) {
        _showMessage('Please select both locations.');
        return;
    }

    if (pickupLocation == dropLocation) {
        _showMessage('Pickup and destination must be different.');

    }

    try {

    } finally {

    if (picked != null) setState(() => _returnDate = picked);
  }

  Future<void> _selectReturnTime() async {
    final picked = await showTimePicker(
      context: context,
      initialTime: _returnTime ?? TimeOfDay.now(),

  }

  String _dateLabel(DateTime? value) {
    if (value == null) return 'Select date';
    return '${value.day.toString().padLeft(2, '0')}/${value.month.toString().padLeft(2, '0')}/${value.year}';
  }

  String _timeLabel(TimeOfDay? value) =>
      value == null ? 'Select time' : value.format(context);

  int? _readPassengerId(Map<String, dynamic> user) {
    final candidates = [user['id'], user['user_id'], user['passenger_id']];
    for (final value in candidates) {
      if (value is int) return value;
      if (value is String) {
        final parsed = int.tryParse(value);
        if (parsed != null) return parsed;
      }
    }
    return null;
  }

  Future<void> _submitBooking() async {
    final scheduledAt = _combineDateAndTime(_travelDate, _travelTime);
    if (scheduledAt == null) {
      setState(() => _errorMessage = 'Please select your travel date and time.');
      return;
    }
    if (scheduledAt.isBefore(DateTime.now())) {
      setState(() => _errorMessage = 'Travel date and time must be in the future.');
      return;
    }

    DateTime? returnAt;
    if (_isRoundTrip) {
      returnAt = _combineDateAndTime(_returnDate, _returnTime);
      if (returnAt == null) {
        setState(() => _errorMessage = 'Please select your return date and time.');
        return;
      }
      if (!returnAt.isAfter(scheduledAt)) {
        setState(() => _errorMessage = 'Return date and time must be after departure.');
        return;
      }
    }

    setState(() {
      _isSubmitting = true;
      _errorMessage = null;
    });

    try {
      final user = await ApiService.getCurrentUser();
      final passengerId = _readPassengerId(user);
      if (passengerId == null) {
        throw Exception('Could not find your passenger ID in the login profile. Please share the /auth/me response with support.');
      }

      final result = await ApiService.createBooking(
        passengerId: passengerId,
        tripType: widget.tripType,
        preferredVehicleType: widget.vehicleModel,
        pickupCity: widget.pickupCity,
        dropCity: widget.dropCity,
        pickupAddress: widget.pickupAddress,
        dropAddress: widget.dropAddress,
        distanceKm: widget.distanceKm,
        scheduledAt: scheduledAt,
        returnAt: returnAt,
        passengerCount: _passengerCount,
        specialInstructions: _instructionsController.text.trim().isEmpty
            ? null
            : _instructionsController.text.trim(),
      );

      if (!mounted) return;
      setState(() {
        _bookingResult = result;
        _isSubmitting = false;
      });
    } catch (e) {
      if (!mounted) return;
      setState(() {
        _errorMessage = e.toString().replaceFirst('Exception: ', '');
        _isSubmitting = false;
      });
    }
  }

class _MyBookingsScreenState extends State<MyBookingsScreen> {
  bool isLoading = true;
  String? errorMessage;
  List<Map<String, dynamic>> bookings = [];
  String selectedFilter = 'All';
  DateTime? selectedDate;

  static const List<String> filters = [
    'All',
    'Pending',
    'Active',
    'Completed',
    'Cancelled',
  ];

  @override
  void initState() {
    super.initState();
    _loadBookings();
  }

  Future<void> _loadBookings() async {
    setState(() {
      isLoading = true;
      errorMessage = null;
    });

    try {
      final result = await ApiService.getMyBookings();

      if (!mounted) return;

      setState(() {
        bookings = result;
        isLoading = false;
      });
    } catch (e) {
      if (!mounted) return;

      setState(() {
        errorMessage = e.toString().replaceFirst('Exception: ', '');
        isLoading = false;
      });
    }
  }

  Future<void> _pickDate() async {
    final picked = await showDatePicker(
      context: context,
      initialDate: selectedDate ?? DateTime.now(),
      firstDate: DateTime(2020),
      lastDate: DateTime(2100),
      helpText: 'Select booking date',
    );

    if (picked != null && mounted) {
      setState(() {
        selectedDate = picked;
      });
    }
  }

  void _clearDateFilter() {
    setState(() {
      selectedDate = null;
    });
  }

  String _status(Map<String, dynamic> booking) {
    return (booking['booking_status'] ?? '').toString().toLowerCase();
  }

  List<Map<String, dynamic>> get filteredBookings {
    List<Map<String, dynamic>> result;

    switch (selectedFilter) {
      case 'Pending':
        result = bookings.where((b) => _status(b) == 'pending').toList();
        break;

      case 'Active':
        result = bookings.where((b) {
          return [
            'accepted',
            'driver_assigned',
            'ongoing',
          ].contains(_status(b));
        }).toList();
        break;

      case 'Completed':
        result = bookings.where((b) => _status(b) == 'completed').toList();
        break;

      case 'Cancelled':
        result = bookings.where((b) => _status(b) == 'cancelled').toList();
        break;

      default:
        result = List<Map<String, dynamic>>.from(bookings);
    }

    // Apply selected date filter.
    if (selectedDate != null) {
      result = result.where((booking) {
        final date = DateTime.tryParse(
          booking['scheduled_at']?.toString() ?? '',
        )?.toLocal();

        return date != null &&
            date.year == selectedDate!.year &&
            date.month == selectedDate!.month &&
            date.day == selectedDate!.day;
      }).toList();
    }

    return result;
  }

  Color _statusColor(String status) {
    switch (status) {
      case 'pending':
        return Colors.orange;

      case 'accepted':
      case 'driver_assigned':
        return Colors.blue;

      case 'ongoing':
        return Colors.green;

      case 'completed':
        return Colors.teal;

      case 'cancelled':
        return Colors.red;

      default:
        return AppColors.textGrey;
    }
  }

  String _prettyStatus(String status) {
    if (status.isEmpty) return 'Unknown';

    return status
        .split('_')
        .map(
          (part) => part.isEmpty
              ? part
              : '${part[0].toUpperCase()}${part.substring(1)}',
        )
        .join(' ');
  }

  String _formatDate(dynamic value) {
    if (value == null) return 'Not scheduled';

    final date = DateTime.tryParse(value.toString())?.toLocal();

    if (date == null) return 'Not scheduled';

    const months = [
      'Jan',
      'Feb',

class _BookingDetailScreenState extends State<BookingDetailScreen> {
  bool isCancelling = false;
  bool isLoadingReview = true;
  bool isSubmittingReview = false;

  Map<String, dynamic>? existingReview;
  String? reviewLoadError;

  int selectedRating = 0;

  final TextEditingController reviewController = TextEditingController();
  @override
  void initState() {
    super.initState();

    final status = (widget.booking['booking_status'] ?? '')
        .toString()
        .toLowerCase();

    debugPrint('â­ Review initialization: $status');

    if (status == 'completed') {
      _loadExistingReview();
    } else {
      isLoadingReview = false;
    }
  }
  Map<String, dynamic> get route {
    final value = widget.booking['route'];

    if (value is Map) {
      return Map<String, dynamic>.from(value);
    }

    return {};
  }

  void dispose() {
    reviewController.dispose();
    super.dispose();
  }

  String _formatDate(dynamic value) {
    if (value == null) return 'Not scheduled';

    final date = DateTime.tryParse(value.toString())?.toLocal();

    if (date == null) return 'Not scheduled';

    const months = [
      'January',
      'February',
      'March',
      'April',
      'May',
      'June',
      'July',
      'August',
      'September',
      'October',
      'November',
      'December',
    ];

    final hour = date.hour % 12 == 0 ? 12 : date.hour % 12;
    final minute = date.minute.toString().padLeft(2, '0');
    final period = date.hour >= 12 ? 'PM' : 'AM';

    return '${date.day} ${months[date.month - 1]} ${date.year}, '
        '$hour:$minute $period';
  }

  String _money(dynamic value) {
    final amount = double.tryParse(value?.toString() ?? '');

    return amount == null ? 'â€”' : 'â‚¹${amount.toStringAsFixed(2)}';
  }

  Future<void> _cancelBooking() async {
    final shouldCancel = await showDialog<bool>(
      context: context,
      builder: (dialogContext) => AlertDialog(
        title: const Text('Cancel booking?'),
        content: const Text(
          'Are you sure you want to cancel this booking? '
          'This action cannot be undone.',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(dialogContext, false),
            child: const Text('Keep booking'),
          ),
          ElevatedButton(
            onPressed: () => Navigator.pop(dialogContext, true),
            style: ElevatedButton.styleFrom(backgroundColor: Colors.red),
            child: const Text('Cancel booking'),
          ),
        ],
      ),
    );

    if (shouldCancel != true || !mounted) return;

    final bookingId = int.tryParse(widget.booking['id'].toString());

    if (bookingId == null) {
      ScaffoldMessenger.of(
        context,
      ).showSnackBar(const SnackBar(content: Text('Invalid booking ID')));
      return;
    }

    setState(() {
      isCancelling = true;
    });

    try {
      await ApiService.cancelBooking(bookingId: bookingId);

      if (!mounted) return;

      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Booking cancelled successfully')),
      );

      Navigator.pop(context, true);
    } catch (e) {
      if (!mounted) return;

      setState(() {
        isCancelling = false;
      });

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text(e.toString().replaceFirst('Exception: ', ''))),
      );
    }
  }

  Future<void> _loadExistingReview() async {
    final bookingId = int.tryParse(widget.booking['id']?.toString() ?? '');

    if (bookingId == null) {
      setState(() {
        isLoadingReview = false;
        reviewLoadError = 'Invalid booking ID';
      });
      return;
    }

    try {
      final result = await ApiService.getBookingReview(bookingId: bookingId);

      if (!mounted) return;

      setState(() {
        existingReview = result;
        selectedRating = result == null
            ? 0
            : int.tryParse(result['rating'].toString()) ?? 0;
        reviewController.text = result?['review']?.toString() ?? '';
        isLoadingReview = false;
        reviewLoadError = null;
      });
    } catch (e) {
      if (!mounted) return;

      setState(() {
        isLoadingReview = false;
        reviewLoadError = e.toString().replaceFirst('Exception: ', '');
      });
    }
  }

  Future<void> _submitReview() async {
    if (selectedRating < 1 || selectedRating > 5) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Please select a star rating')),
      );
      return;
    }

    final bookingId = int.tryParse(widget.booking['id']?.toString() ?? '');

    if (bookingId == null) return;

    setState(() {
      isSubmittingReview = true;
    });

    try {
      final result = await ApiService.submitDriverReview(
        bookingId: bookingId,
        rating: selectedRating,
        review: reviewController.text.trim().isEmpty
            ? null
            : reviewController.text.trim(),
      );

      if (!mounted) return;

      setState(() {
        existingReview = result;
        isSubmittingReview = false;
      });

      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Thank you! Your review was submitted.')),
      );
    } catch (e) {
      if (!mounted) return;

      setState(() {
        isSubmittingReview = false;
      });

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text(e.toString().replaceFirst('Exception: ', ''))),
      );
    }
  }
