  Future<void> _setDestination(
    LatLng location, {
    bool moveCamera = false,
    String? knownAddress,
  }) async {
    setState(() {
      _destination = location;
      if (knownAddress != null) {
        _destinationAddress = knownAddress;
      }
    });

    if (moveCamera) {
      await _mapController?.animateCamera(
        CameraUpdate.newLatLngZoom(location, 14),
      );
    }

    if (knownAddress == null) {
      await _reverseGeocode(location);
    }
  }

  Future<void> _reverseGeocode(LatLng location) async {
    try {
      final places = await placemarkFromCoordinates(
        location.latitude,
        location.longitude,
      );

      if (places.isEmpty || !mounted) return;

      final place = places.first;
      final parts = [
        place.name,
        place.subLocality,
        place.locality,
      ].where((value) => value != null && value.isNotEmpty).toList();

      setState(() {
        _destinationAddress = parts.isEmpty
            ? 'Selected Location'
            : parts.join(', ');
        _destinationCity = place.locality ??
            place.subAdministrativeArea ??
            'Ujjain';
      });
    } catch (_) {
      if (mounted) {
        setState(() => _destinationAddress = 'Selected Location');
      }
    }
  }

  Future<void> _searchDestination() async {
    final query = _searchController.text.trim();
    if (query.isEmpty) return;

    try {
      final results = await locationFromAddress(query);

      if (results.isEmpty) {
        _message('Destination not found.');
        return;
      }

      await _setDestination(
        LatLng(
          results.first.latitude,
          results.first.longitude,
        ),
        moveCamera: true,
        knownAddress: query,
      );
    } catch (_) {
      _message('Unable to search this destination.');
    }
  }

  Future<void> _useCurrentLocation() async {
    setState(() => _loadingLocation = true);

    try {
      var permission = await Geolocator.checkPermission();

      if (permission == LocationPermission.denied) {
        permission = await Geolocator.requestPermission();
      }

      if (permission == LocationPermission.denied ||
          permission == LocationPermission.deniedForever) {
        _message('Location permission is required.');
        return;
      }

      final position = await Geolocator.getCurrentPosition(
        locationSettings: const LocationSettings(
          accuracy: LocationAccuracy.high,
        ),
      );

      await _setDestination(
        LatLng(position.latitude, position.longitude),
        moveCamera: true,
      );
    } catch (_) {
      _message('Unable to get current location.');
    } finally {
      if (mounted) {
        setState(() => _loadingLocation = false);
      }
    }
  }

  void _selectPopular(
    String name,
    String city,
    double latitude,
    double longitude,

    _searchController.text = name;
    setState(() {

  }

  
  Future<void> _confirmDestination() async {
    // STEP 1: Open Trip Type screen
    final tripResult = await Navigator.push<Map<String, dynamic>>(
      context,
      MaterialPageRoute(
        builder: (_) => const TripTypeScreen(),
      ),
    );

    if (tripResult == null || !mounted) return;

    // STEP 2: Calculate actual route distance
    double distanceKm;
    bool usingFallbackDistance = false;

    try {
      distanceKm = await ApiService.getRouteDistanceKm(
        pickupLatitude: widget.pickupLatitude,
        pickupLongitude: widget.pickupLongitude,
        dropLatitude: _destination.latitude,
        dropLongitude: _destination.longitude,
      );

      // Validate API response
      if (!distanceKm.isFinite || distanceKm <= 0) {
        throw Exception('Invalid distance returned by API');
      }

      debugPrint('API route distance: $distanceKm km');
    } catch (e) {
      debugPrint('Route distance API failed: $e');

      // Use temporary fallback distance
      distanceKm = 100.0;
      usingFallbackDistance = true;

      debugPrint('Using fallback distance: 100 km');
    }

    if (!mounted) return;

    // Inform passenger when fallback is being used
    if (usingFallbackDistance) {
      _message(
        'Route distance unavailable. Temporary estimate: 100 km.',
      );
    }

    // STEP 3: Open Vehicle Selection screen
    final vehicleResult =
        await Navigator.push<Map<String, dynamic>>(
      context,
      MaterialPageRoute(
        builder: (_) => VehicleSelectionScreen(
          pickupCity: widget.pickupCity,
          dropCity: _destinationCity,
          tripType: tripResult['trip_type'],
          distanceKm: distanceKm,
        ),
      ),
    );

    if (vehicleResult == null || !mounted) return;

    // STEP 4: Return all booking details
    Navigator.pop(context, {
      'pickup_latitude': widget.pickupLatitude,
      'pickup_longitude': widget.pickupLongitude,
      'pickup_address': widget.pickupAddress,
      'pickup_city': widget.pickupCity,

      'drop_latitude': _destination.latitude,
      'drop_longitude': _destination.longitude,
      'drop_address': _destinationAddress,
      'drop_city': _destinationCity,

      'distance_km': distanceKm,
      'using_fallback_distance': usingFallbackDistance,

      ...tripResult,
      ...vehicleResult,
    });
  }

    setState(() {
      isLoading = true;
      errorMessage = null;
    });

    try {
      final result = await ApiService.estimateFare(
        tripType: widget.tripType,
        vehicleType: widget.vehicleModel,
        distanceKm: widget.distanceKm,
        passengerCount: 1,
      );

        isLoading = false;
      });
    } catch (e) {

        errorMessage =

        isLoading = false;
      });
    }

    if (date == null || time == null) return null;
    return DateTime(date.year, date.month, date.day, time.hour, time.minute);
  }

    final picked = await showDatePicker(

      initialDate: _travelDate ?? now,

    if (value == null) return 'Select date';
    return '${value.day.toString().padLeft(2, '0')}/${value.month.toString().padLeft(2, '0')}/${value.year}';
  }

  String _timeLabel(TimeOfDay? value) =>
      value == null ? 'Select time' : value.format(context);

  int? _readPassengerId(Map<String, dynamic> user) {
    final candidates = [user['id'], user['user_id'], user['passenger_id']];
    for (final value in candidates) {
      if (value is int) return value;
      if (value is String) {
        final parsed = int.tryParse(value);
        if (parsed != null) return parsed;
      }
    }
    return null;
  }

  Future<void> _submitBooking() async {
    final scheduledAt = _combineDateAndTime(_travelDate, _travelTime);
    if (scheduledAt == null) {
      setState(() => _errorMessage = 'Please select your travel date and time.');
      return;
    }
    if (scheduledAt.isBefore(DateTime.now())) {
      setState(() => _errorMessage = 'Travel date and time must be in the future.');
      return;
    }

    DateTime? returnAt;
    if (_isRoundTrip) {
      returnAt = _combineDateAndTime(_returnDate, _returnTime);
      if (returnAt == null) {
        setState(() => _errorMessage = 'Please select your return date and time.');
        return;
      }
      if (!returnAt.isAfter(scheduledAt)) {
        setState(() => _errorMessage = 'Return date and time must be after departure.');
        return;
      }
    }

    setState(() {
      _isSubmitting = true;
      _errorMessage = null;
    });

    try {
      final user = await ApiService.getCurrentUser();
      final passengerId = _readPassengerId(user);
      if (passengerId == null) {
        throw Exception('Could not find your passenger ID in the login profile. Please share the /auth/me response with support.');
      }

      final result = await ApiService.createBooking(
        passengerId: passengerId,
        tripType: widget.tripType,
        preferredVehicleType: widget.vehicleModel,
        pickupCity: widget.pickupCity,
        dropCity: widget.dropCity,
        pickupAddress: widget.pickupAddress,
        dropAddress: widget.dropAddress,
        distanceKm: widget.distanceKm,
        scheduledAt: scheduledAt,
        returnAt: returnAt,
        passengerCount: _passengerCount,
        specialInstructions: _instructionsController.text.trim().isEmpty

      setState(() {
        _errorMessage = e.toString().replaceFirst('Exception: ', '');
        _isSubmitting = false;

import 'dart:convert';
import 'package:http/http.dart' as http;
import 'package:shared_preferences/shared_preferences.dart';

class ApiService {
  // Physical phone: use your computer's local IPv4 address.
  // Android emulator: use 10.0.2.2 instead of localhost.
  static const String baseUrl = 'http://192.168.29.201:8000';

  static Future<Map<String, dynamic>> login({
    required String phone,
    required String password,
  }) async {
    final response = await http.post(
      Uri.parse('$baseUrl/auth/login'),
      headers: {'Content-Type': 'application/json'},
      body: jsonEncode({
        'phone': phone,
        'password': password,
      }),
    );

    final data = jsonDecode(response.body);

    if (response.statusCode == 200) {
      final prefs = await SharedPreferences.getInstance();
      await prefs.setString('access_token', data['access_token']);
      return data;
    }

    throw Exception(data['detail'] ?? 'Login failed');
  }

  static Future<Map<String, dynamic>> register({
    required String fullName,
    required String phone,
    required String password,
  }) async {
    final response = await http.post(
      Uri.parse('$baseUrl/auth/register'),
      headers: {'Content-Type': 'application/json'},
      body: jsonEncode({
        'full_name': fullName,
        'phone': phone,
        'password': password,
      }),
    );

    final data = jsonDecode(response.body);

    if (response.statusCode == 200 || response.statusCode == 201) {
      final prefs = await SharedPreferences.getInstance();
      await prefs.setString('access_token', data['access_token']);
      return data;
    }

    throw Exception(data['detail'] ?? 'Registration failed');
  }

  static Future<void> logout() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.remove('access_token');
  }

  static Future<String?> getToken() async {
    final prefs = await SharedPreferences.getInstance();
    return prefs.getString('access_token');
  }

  static Future<Map<String, dynamic>> getCurrentUser() async {
    final token = await getToken();

    if (token == null || token.isEmpty) {
      throw Exception('No login session found');
    }

    final response = await http.get(
      Uri.parse('$baseUrl/auth/me'),
      headers: {
        'Authorization': 'Bearer $token',
        'Content-Type': 'application/json',
      },
    );

    if (response.statusCode == 200) {
      return jsonDecode(response.body) as Map<String, dynamic>;
    }

    if (response.statusCode == 401) {
      await logout();
      throw Exception('Session expired. Please login again.');
    }

    throw Exception('Unable to verify login session');
  }

  // Estimate ride fare.
  static Future<Map<String, dynamic>> estimateFare({
    required String tripType,
    required String vehicleType,
    required double distanceKm,
    int passengerCount = 1,
  }) async {
    final token = await getToken();

    if (token == null || token.isEmpty) {

    );

  // Register Firebase device token with backend.
  static Future<void> registerDeviceToken(String fcmToken) async {
    final token = await getToken();

    if (token == null || token.isEmpty) {
      throw Exception('Please login first');
    }

    final response = await http.post(
      Uri.parse('$baseUrl/notifications/device-token'),
      headers: {
        'Authorization': 'Bearer $token',
        'Content-Type': 'application/json',
      },
      body: jsonEncode({
        'token': fcmToken,
        'platform': 'android',
      }),
    );

    if (response.statusCode != 200) {
      throw Exception('Device token registration failed: ${response.body}');
    }
  }

  // Remove Firebase device token on logout.
  static Future<void> removeDeviceToken(String fcmToken) async {
    final token = await getToken();

    if (token == null || token.isEmpty) return;

    final uri = Uri.parse('$baseUrl/notifications/device-token')
        .replace(queryParameters: {'token': fcmToken});

    final response = await http.delete(
      uri,
      headers: {
        'Authorization': 'Bearer $token',
      },
    );

    if (response.statusCode != 200) {
      throw Exception('Device token removal failed: ${response.body}');
    }
  }
